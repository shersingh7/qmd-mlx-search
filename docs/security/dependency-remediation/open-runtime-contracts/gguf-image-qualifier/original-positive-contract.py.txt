"""Positive contract held-outs: correct release remains valid with upstream suffixes."""
from __future__ import annotations
import hashlib
import json
from pathlib import Path
import runpy
import subprocess
import sys

import pytest

S = runpy.run_path(str(Path(__file__).with_name('review_support.py')))
PROBE = S['REPO'] / 'finetune/tools/qualify_gguf_image.py'

@pytest.mark.parametrize('label,display,expected', [
    ('bare-control', 'uv 0.11.18', 0),
    ('upstream-without-commit', 'uv 0.11.18 (x86_64-unknown-linux-musl)', 0),
    ('upstream-with-commit', 'uv 0.11.18 (e32666915 2026-06-01 x86_64-unknown-linux-musl)', 0),
    ('wrong-release-control', 'uv 0.11.17 (x86_64-unknown-linux-musl)', 2),
])
def test_uv_release_contract(tmp_path, label, display, expected):
    assert hashlib.sha256(PROBE.read_bytes()).hexdigest() == '6717b3770708789be996da14ef3ce614f81f15c1ce0713ed03aa457201729ecf'
    uv = tmp_path / 'uv'
    uv.write_text('#!/bin/sh\nprintf "%s\\n" "' + display + '"\n')
    uv.chmod(0o700)
    env = S['environment']('contract-' + label)
    env['PATH'] = str(tmp_path) + ':' + env['PATH']
    result = subprocess.run([sys.executable, '-I', '-B', '-S', str(PROBE)], env=env, cwd=tmp_path,
                            capture_output=True, text=True, timeout=120)
    receipt = {'label': label, 'synthetic_uv_display': display, 'expected_exit': expected,
               'actual_exit': result.returncode, 'stdout': result.stdout, 'stderr': result.stderr,
               'scope': 'HOST probe with real compiler/make and synthetic uv --version; NOT image execution'}
    (S['ROOT'] / 'evidence' / ('contract-' + label + '.json')).write_text(json.dumps(receipt, indent=2, sort_keys=True))
    assert result.returncode == expected, receipt
    if expected == 0:
        assert json.loads(result.stdout)['outputs'] == ['qmd-c-ok', 'qmd-cxx-ok']
    else:
        assert not result.stdout and 'requires uv 0.11.18' in result.stderr

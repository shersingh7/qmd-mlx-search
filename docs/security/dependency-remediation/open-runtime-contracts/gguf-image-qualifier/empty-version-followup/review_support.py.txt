def digest(content):

def run_case(name, command, *, cwd=None, env=None, timeout=180):
    start = time.monotonic()
    p = subprocess.run([str(x) for x in command], cwd=cwd or ROOT, env=env or environment(name),
                       capture_output=True, text=True, timeout=timeout)
    receipt = {'name': name, 'command': [str(x) for x in command], 'cwd': str(cwd or ROOT),
               'returncode': p.returncode, 'seconds': time.monotonic() - start,
               'stdout': p.stdout, 'stderr': p.stderr}
    (ROOT / 'evidence' / (name + '.json')).write_text(json.dumps(receipt, indent=2, sort_keys=True))
    return receipt

def junit(path):
    # XML is generated locally by the hash-verified pytest suite, not untrusted input.
    tree = ET.parse(path)
    cases = list(tree.iter('testcase'))
    return {'tests': len(cases), 'failures': sum(len(x.findall('failure')) for x in cases),
            'errors': sum(len(x.findall('error')) for x in cases),
            'skipped': sum(len(x.findall('skipped')) for x in cases),
            'case_names': [x.attrib.get('name') for x in cases],
            'failure_cases': [{'name': x.attrib.get('name'), 'message': x.find('failure').attrib.get('message'),
                               'text': x.find('failure').text} for x in cases if x.find('failure') is not None]}

def suite(name, repo=REPO, *, selected=None):
    (ROOT / 'pytest-tmp').mkdir(exist_ok=True)
    xml = ROOT / 'evidence' / (name + '.xml')
    command = [PYTHON, '-I', '-B', '-m', 'pytest', '-c', '/dev/null', '-p', 'no:cacheprovider',
               '--rootdir=' + str(repo), '--confcutdir=' + str(repo / 'finetune/tests'),
               '--basetemp=' + str(ROOT / 'pytest-tmp' / name), '--junitxml=' + str(xml), '-q', '--tb=short']
    command += selected or [repo / 'finetune/tests/test_locked_job_submission.py']
    receipt = run_case(name, command, cwd=repo)
    receipt['junit'] = junit(xml)
    (ROOT / 'evidence' / (name + '.json')).write_text(json.dumps(receipt, indent=2, sort_keys=True))
    return receipt

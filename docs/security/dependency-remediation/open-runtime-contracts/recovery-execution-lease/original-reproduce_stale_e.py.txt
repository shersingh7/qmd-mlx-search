"""Fresh bounded reproduction. Exit 1 asserts the stale-E/EOF contract fails."""
import json, os, pathlib, select, signal, subprocess, tempfile, time
BASE=pathlib.Path(__file__).parent
RUN=pathlib.Path(tempfile.mkdtemp(prefix='stale-e-confirm-',dir=BASE/'probes'))
SOURCE=BASE/'snapshot/scripts/phase5_recovery_process.py'
ENV=json.loads((BASE/'focused-invocation.json').read_text())['env']
observations=[]
for name,grant,held in [('positive','E-open',False),('eof-no-E','eof',False),('stale-E-eof','E-eof',True)]:
    root=RUN/name;root.mkdir();work=root/'work';wrapper=root/'supervisor-wrapper.py'
    wrapper.write_text('''import json,os,sys,time
os.dup2(1,3)
source=%r
sys.path.insert(0,os.path.dirname(source))
scope=dict(__name__='fresh_stale_E_repro')
exec(compile(open(source,'rb').read(),source,'exec'),scope)
original_read=os.read
original_write=os.write
held=%r
def record(stage):
    with open(%r,'a') as out: out.write(json.dumps(dict(stage=stage,pid=os.getpid(),time=time.monotonic()))+'\\n')
def read(fd,n):
    data=original_read(fd,n)
    if held and fd==0 and data==b'E':
        record('E-read');time.sleep(1.300);record('E-resumed')
    return data
def write(fd,data):
    written=original_write(fd,data)
    if held and fd!=3 and data==b'E':
        record('gate-opened');time.sleep(.050)
    return written
os.read=read;os.write=write
scope['main']()
''' % (str(SOURCE),held,str(root/'stages.jsonl')))
    sentinel=subprocess.Popen(['/usr/bin/python3','-B','-S','-c','import time;time.sleep(60)'],env=ENV,
        stdin=subprocess.DEVNULL,stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL,start_new_session=True)
    child=subprocess.Popen(['/usr/bin/python3','-B','-S','-u',str(wrapper),json.dumps(dict(
        command=['/bin/sh','-c','printf actual-user-work > "$1"','owned-shell',str(work)],timeoutMs=200,contained=False))],
        env=ENV,stdin=subprocess.PIPE,stdout=subprocess.PIPE,stderr=subprocess.PIPE,start_new_session=True)
    started=time.monotonic();deadline=started+1.800
    pending=b'';stdout=b'';stderr=b'';messages=[];guard=False;finally_kills=[]
    streams={child.stdout.fileno():'stdout',child.stderr.fileno():'stderr'}
    try:
        while streams:
            remaining=deadline-time.monotonic()
            if remaining<=0: guard=True;break
            ready,_,_=select.select(list(streams),[],[],remaining)
            if not ready: guard=True;break
            for fd in ready:
                data=os.read(fd,65536)
                if not data: del streams[fd];continue
                if fd==child.stderr.fileno(): stderr+=data;continue
                stdout+=data;pending+=data
                while b'\n' in pending:
                    line,pending=pending.split(b'\n',1);message=json.loads(line);messages.append(message)
                    if message['type']=='prepared': child.stdin.write(b'S');child.stdin.flush()
                    if message['type']=='ready':
                        if grant.startswith('E'): child.stdin.write(b'E');child.stdin.flush()
                        if grant!='E-open':child.stdin.close()
        if not guard:child.wait(timeout=max(.001,deadline-time.monotonic()))
        def alive(pid):
            try:os.kill(pid,0);return True
            except ProcessLookupError:return False
        workers=[message['pid'] for message in messages if message['type']=='ready']
        record=dict(name=name,grant=grant,held=held,executionDeadlineMs=200,guardMs=1800,guardFired=guard,
            elapsedMs=(time.monotonic()-started)*1000,supervisorPid=child.pid,supervisorAbsent=not alive(child.pid),
            workersAbsent=all(not alive(pid) for pid in workers),sentinelAlive=sentinel.poll() is None,
            work=work.read_text() if work.exists() else None,messages=messages,stderr=stderr.decode(errors='replace'))
        (root/'before-finally.json').write_text(json.dumps(record,indent=2)+'\n');observations.append(record)
        (root/'stdout.log').write_bytes(stdout);(root/'stderr.log').write_bytes(stderr)
        print(json.dumps(record),flush=True)
    finally:
        if child.poll() is None: finally_kills.append(child.pid);child.kill();child.wait()
        sentinel.kill();sentinel.wait()
        (root/'finally.json').write_text(json.dumps(dict(fixtureKills=finally_kills,sentinelAbsent=sentinel.poll() is not None),indent=2)+'\n')
(RUN/'results.json').write_text(json.dumps(observations,indent=2)+'\n')
print('Fresh reproduction artifacts: '+str(RUN),flush=True)
assert observations[0]['work']=='actual-user-work', 'positive control must really execute'
assert observations[1]['work'] is None, 'EOF without execution admission must remain gated'
assert observations[2]['work'] is None, 'SPEC_FAIL: E consumed before a scheduling hold was forwarded after deadline and queued EOF; actual user work ran'

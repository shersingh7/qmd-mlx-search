class Refused(Exception):
    def __init__(self, message, code=1):
        super().__init__(message)
        self.code = (128 - code) if -127 <= code < 0 else (code if 0 < code < 256 else 1)

def require(condition, message):
    if not condition:
        raise Refused(message)

def read_regular(path):
    # No symlink ancestors/final leaf, hardlinks or special files; bounded read.
    require(path.is_absolute() and '..' not in path.parts, 'unsafe source path')
    directory = os.open(path.anchor, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
    try:
        for part in path.parts[1:-1]:
            child = os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=directory)
            os.close(directory)
            directory = child
        fd = os.open(path.name, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK, dir_fd=directory)
        with os.fdopen(fd, 'rb') as stream:
            info = os.fstat(stream.fileno())
            require(stat.S_ISREG(info.st_mode) and info.st_nlink == 1 and info.st_size <= 1048576,
                    'unsafe/nonregular/oversize public file')
            raw = stream.read(1048577)
            require(len(raw) <= 1048576, 'oversize public file')
            return raw
    finally:
        os.close(directory)

def unique_object(pairs):
    value = {}
    for key, item in pairs:
        require(key not in value, 'duplicate JSON key')
        value[key] = item
    return value

def parse_json(text):
    return json.loads(text, object_pairs_hook=unique_object)

def invoke(name, argv, seconds, check=True):
    # File-backed output avoids pipe/descendant EOF hangs. Every CLI gets a new process group.
    print(json.dumps({'start': name, 'argv': argv, 'timeout_seconds': seconds}), flush=True)
    interrupted = None

        child = subprocess.Popen(argv, cwd=owned, stdout=out, stderr=err, start_new_session=True)
        try:
            code = child.wait(timeout=seconds)
        except BaseException as exc:
            for sig in (signal.SIGTERM, signal.SIGKILL):
                try:
                    os.killpg(child.pid, sig)
                except ProcessLookupError:
                    pass
                try:
                    child.wait(timeout=5)
                    break
                except subprocess.TimeoutExpired:
                    pass
            code = 124 if isinstance(exc, subprocess.TimeoutExpired) else (exc.code if isinstance(exc, Refused) else 1)
            if not isinstance(exc, subprocess.TimeoutExpired):
                interrupted = exc
    stdout = (owned / (name + '.stdout')).read_text(errors='replace')
    stderr = (owned / (name + '.stderr')).read_text(errors='replace')
    record = {'name': name, 'argv': argv, 'timeout_seconds': seconds, 'exit': code,
              'process_exit': child.returncode, 'stdout': stdout, 'stderr': stderr}

    # JSON escaping prevents command output becoming GitHub workflow commands.

    if name == 'probe':
        report['probe_exit'] = code
    if interrupted is not None:
        raise interrupted
    if check and code != 0:
        raise Refused(name + ' failed', code)
    return code, stdout

def container_ids(name):
    _, raw = invoke(name, docker + ['container', 'ls', '--all', '--no-trunc', '--format', '{{json .ID}}'], 15)
    ids = [parse_json(line) for line in raw.splitlines()]
    require(all(isinstance(cid, str) and re.fullmatch('[0-9a-f]{64}', cid) for cid in ids),
            'malformed container-list readback')
    return ids

    require(cid not in container_ids('cleanup-after'), 'owned container still present')
    report['cleanup'].update(status='absence-confirmed', absence_verified=True)

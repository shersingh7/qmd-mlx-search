function calls(source, names) {
  const file = ts.createSourceFile("input.ts", source, ts.ScriptTarget.Latest, true, ts.ScriptKind.TS);

    if (ts.isCallExpression(node) && names.includes(node.expression.getText(file))) result.push(node.getText(file));
    ts.forEachChild(node,visit);
  }
  visit(file); return result;
}

// Explicit bounded TS source surface: exclude historical doc .mjs scripts with
// absolute live-source imports. This is not canonical root typecheck acceptance.
function boundedTypes(which) {
  const base=path.join(root,which);
  const config=ts.readConfigFile(path.join(base,"tsconfig.json"),ts.sys.readFile);
  const parsed=ts.parseJsonConfigFileContent(config.config,ts.sys,base);
  const entries=parsed.fileNames.filter(name=>name.endsWith(".ts"));
  const host=ts.createCompilerHost(parsed.options);
  const getSource=host.getSourceFile.bind(host);
  const readPaths=[];
  host.getSourceFile=(name,...args)=> {
    const absolute=path.resolve(name);
    assert.ok(absolute.startsWith(base+path.sep)||absolute.includes(path.sep+"node_modules"+path.sep),`outside bounded public source closure: ${name}`);
    readPaths.push(absolute); return getSource(name,...args);
  };
  const program=ts.createProgram(entries,parsed.options,host);
  const diagnostics=ts.getPreEmitDiagnostics(program).map(d=>({path:d.file?path.relative(base,d.file.fileName):null,
    code:d.code, start:d.start??null, message:ts.flattenDiagnosticMessageText(d.messageText,"\n")}));
  return { entryCount:entries.length, sourceFilesRead:readPaths, diagnostics };
}

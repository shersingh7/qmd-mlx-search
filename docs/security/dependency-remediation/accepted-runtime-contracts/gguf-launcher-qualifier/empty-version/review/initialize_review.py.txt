def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def record(path, data):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(data, indent=2, sort_keys=True) + '\n')

def files(root):
    result = {}
    for path in root.rglob('*'):
        assert not path.is_symlink(), str(path)
        if path.is_file():
            result[str(path.relative_to(root))] = sha(path)
    return result

def junit(path):
    # These are hash-preserved local pytest receipts, not untrusted XML input.
    cases = list(ET.parse(path).iter('testcase'))
    return {'tests': len(cases), 'failures': sum(len(c.findall('failure')) for c in cases), 'errors': sum(len(c.findall('error')) for c in cases), 'skipped': sum(len(c.findall('skipped')) for c in cases), 'groups': {name: sum(c.get('classname') == name for c in cases) for name in sorted(set(c.get('classname') for c in cases))}, 'failure_texts': [c.find('failure').text for c in cases if c.find('failure') is not None]}

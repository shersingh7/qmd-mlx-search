def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()

def save(name, obj):
    (ROOT / 'evidence' / name).write_text(json.dumps(obj, indent=2, sort_keys=True))

    shutil.copyfile(ROOT / 'repo' / relative,target)
for label,patch,options in [
    ('compose-reverse-repair-check',ROOT/'repair.patch',['--reverse','--check']),
    ('compose-reverse-repair',ROOT/'repair.patch',['--reverse']),

# Inspect actual local call sites as AST/data only, never execute their modules.
callers={}
for relative in ['finetune/convert_gguf.py','finetune/train.py','finetune/train_unsloth.py',
                 'finetune/eval.py','finetune/gguf_toolchain.py']:
    f=ROOT/'repo'/relative
    tree=ast.parse(f.read_bytes(),filename=str(f))
    imports=[{'line':n.lineno,'module':n.module,'names':[a.name for a in n.names]}
             for n in ast.walk(tree) if isinstance(n,ast.ImportFrom)]
    calls=[{'line':n.lineno,'name':n.func.id} for n in ast.walk(tree)
           if isinstance(n,ast.Call) and isinstance(n.func,ast.Name)
           and n.func.id in {'prepare_toolchain','convert_model','check_prerequisites','export_gguf','export_gguf_artifacts','run_eval'}]
    callers[relative]={'sha256':sha(f),'imports':imports,'relevant_calls':calls,'executed':False}
save('actual-callers-as-data.json',callers)

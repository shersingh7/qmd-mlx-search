def dump(path, data):
    path.write_text(json.dumps(data, indent=2) + '\n')

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

records = []
for line in (ROOT / 'focused-tests.log').read_text().splitlines():
    if line.startswith('{"evidence":'):
        records.append(json.loads(line))
dump(ROOT / 'focused-observations.json', records)
counts = Counter(item['evidence'] for item in records)
raw = [item for item in records if item['evidence'] == 'recovery-execution-admission-before-finally']
frames = [item for item in records if item['evidence'] == 'recovery-budget-frame-before-finally']
hosts = [item for item in records if item['evidence'] == 'recovery-parent-death-before-finally']
new_finally = [item for item in records if item['evidence'] in ['recovery-execution-admission-finally', 'recovery-budget-frame-finally', 'recovery-parent-death-finally']]
assert len(raw) == 15 and len(frames) == 7 and len(hosts) == 22 and len(new_finally) == 44
assert all(not item.get('finallyKilled') for item in new_finally)
assert all(item['supervisorAbsent'] and item['workerAbsent'] and item['sentinelAlive'] and not item['guardFired'] for item in raw)
assert all(item['supervisorAbsent'] and not item['forked'] and item['sentinelAlive'] and not item['guardFired'] for item in frames)
assert all(item['supervisorsAbsent'] and item['workersAbsent'] and item['sentinelAlive'] and not item['guardFired'] for item in hosts)
helper_times = [item['helper']['elapsedMs'] for item in hosts if item['mode'] == 'alive']
runtime_summary = {}
for runtime in ['node', 'bun']:

    modes = Counter(item['mode'] for item in samples)
    runtimes = {json.dumps({'executable': item['nativeRuntime']['executable'], 'node': item['nativeRuntime']['versions']['node'], 'bun': item['nativeRuntime']['versions'].get('bun')}, sort_keys=True) for item in samples}
    assert modes == {'valid': 1, 'alive': 3, 'graceful': 3, 'hard': 4}
    runtime_summary[runtime] = {'cases': len(samples), 'modes': dict(modes), 'actualRuntime': [json.loads(x) for x in sorted(runtimes)]}
kills = [item for item in records if item.get('finallyKilled')]
assert len(kills) == 4, kills
assert sorted(item['mode'] for item in kills) == ['escaped-pipe', 'escaped-pipe', 'new-session', 'new-session']

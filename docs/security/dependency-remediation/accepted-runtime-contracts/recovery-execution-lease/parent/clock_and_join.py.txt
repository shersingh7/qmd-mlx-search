"""Actual fork-clock, gate-FD, and kernel join-refusal controls."""
import hashlib
import json
import os
from pathlib import Path
import signal
import subprocess
import sys
import time

ROOT = Path(__file__).resolve().parent
ENV = json.loads((ROOT / 'environment.json').read_text())
SOURCE = ROOT / 'mirror/scripts/phase5_recovery_process.py'

def dump(path, value):
    path.write_text(json.dumps(value, indent=2) + '\n')

clock_code = '''import json,os,sys,time
before=time.monotonic()
r,w=os.pipe()
pid=os.fork()
if pid==0:
 os.close(r);os.write(w,json.dumps(dict(childPid=os.getpid(),value=time.monotonic())).encode());os._exit(0)
os.close(w);child=json.loads(os.read(r,4096));after=time.monotonic();os.waitpid(pid,0)
result=dict(parentPid=os.getpid(),before=before,after=after,child=child,executable=sys.executable,version=sys.version,clock=str(time.get_clock_info('monotonic')))
result['sameInheritedEpoch']=before<=child['value']<=after
print(json.dumps(result));assert result['sameInheritedEpoch']
'''
(ROOT / 'fork-clock.py').write_text(clock_code)
clocks = []
for index in range(3):
    result = subprocess.run(['/usr/bin/python3', '-B', '-S', str(ROOT / 'fork-clock.py')], cwd=ROOT, env=ENV, capture_output=True, text=True, check=True)
    record = json.loads(result.stdout)
    record['iteration'] = index
    clocks.append(record)
dump(ROOT / 'fork-clock-results.json', clocks)

# Pure gate calls retain the original function body and use real kernel pipe
# behavior; no function, time, read-result or cancellation response is faked.
sys.path.insert(0, str(SOURCE.parent))
scope = {'__name__': 'independent_gate_controls'}
exec(compile(SOURCE.read_bytes(), str(SOURCE), 'exec'), scope)
gates = []
for mode in ['live-empty', 'EOF', 'K', 'T', 'unexpected', 'EBADF', 'expired']:
    r, w = os.pipe()
    os.set_blocking(r, False)
    duplicate = os.dup(r)
    os.close(r)
    non_inheritable = not os.get_inheritable(duplicate)
    if mode == 'EOF':
        os.close(w); w = None
    elif mode in ['K', 'T', 'unexpected']:
        os.write(w, {'K': b'K', 'T': b'T', 'unexpected': b'X'}[mode])
    elif mode == 'EBADF':
        os.close(duplicate)
    deadline = time.monotonic() + (-.001 if mode == 'expired' else 1)
    result = scope['execution_allowed'](duplicate, deadline)
    gates.append({'mode': mode, 'allowed': result, 'CLOEXEC': non_inheritable})
    assert result == (mode == 'live-empty') and non_inheritable
    if mode != 'EBADF': os.close(duplicate)
    if w is not None: os.close(w)
dump(ROOT / 'gate-controls.json', {'sourceSha256': hashlib.sha256(SOURCE.read_bytes()).hexdigest(), 'cases': gates})

# A real cross-session kernel join failure must authorize no killpg attempt.
join_root = ROOT / 'join-refusal'
join_root.mkdir()
sentinel = subprocess.Popen(['/usr/bin/python3', '-B', '-S', '-c', 'import time;time.sleep(60)'], cwd=join_root, env=ENV,
    stdin=subprocess.DEVNULL, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL, start_new_session=True)
join_code = '''import json,os,sys
source=%r
sys.path.insert(0,os.path.dirname(source));scope=dict(__name__='independent_join_refusal')
exec(compile(open(source,'rb').read(),source,'exec'),scope)
root=%r;target=%r
original=os.setpgid
def setpgid(pid,group):
 try: original(pid,group)
 except OSError as error:
  with open(root+'/join.json','w') as out:json.dump(dict(errno=error.errno,group=group,currentSession=os.getsid(0),targetSession=os.getsid(target)),out)
  raise
def killpg(*args):
 with open(root+'/UNAUTHORIZED-SIGNAL-ATTEMPT.json','w') as out:json.dump(args,out)
os.setpgid,os.killpg=setpgid,killpg
scope['refuse_execution'](target)
''' % (str(SOURCE), str(join_root), sentinel.pid)
(join_root / 'join-probe.py').write_text(join_code)
try:
    result = subprocess.run(['/usr/bin/python3', '-B', '-S', str(join_root / 'join-probe.py')], cwd=join_root, env=ENV,
        capture_output=True, text=True, timeout=3, start_new_session=True)
    joined = json.loads((join_root / 'join.json').read_text())
    record = {'exitCode': result.returncode, 'join': joined, 'sentinelPid': sentinel.pid,
        'sentinelAliveBeforeCleanup': sentinel.poll() is None,
        'signalAttempt': (join_root / 'UNAUTHORIZED-SIGNAL-ATTEMPT.json').exists(), 'stderr': result.stderr}
    dump(join_root / 'before-finally.json', record)
    assert result.returncode == 125 and joined['errno'] in (1, 3) and joined['currentSession'] != joined['targetSession']
    assert not record['signalAttempt'] and record['sentinelAliveBeforeCleanup']
finally:
    if sentinel.poll() is None:
        sentinel.kill(); sentinel.wait()
    dump(join_root / 'finally.json', {'sentinelAbsent': sentinel.poll() is not None})
print(json.dumps({'forkClockPasses': len(clocks), 'actualPython': clocks[0]['executable'],
    'gatePasses': len(gates), 'realCrossSessionRefusal': record}, indent=2))

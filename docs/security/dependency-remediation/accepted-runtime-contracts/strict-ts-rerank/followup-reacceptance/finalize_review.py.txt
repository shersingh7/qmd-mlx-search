def digest(p):
    h=hashlib.sha256()
    with Path(p).open('rb') as f:
        for x in iter(lambda:f.read(1048576),b''):h.update(x)
    return h.hexdigest()

def item(p):
    p=Path(p);s=p.lstat()
    if stat.S_ISLNK(s.st_mode):return {'kind':'link','target':os.readlink(p),'mtime_ns':s.st_mtime_ns}
    assert stat.S_ISREG(s.st_mode),str(p)
    return {'kind':'file','size':s.st_size,'mtime_ns':s.st_mtime_ns,'sha256':digest(p)}
def verify(records):
    failures=[]
    for name,expected in records.items():
        expected=dict(expected)
        if 'mtimeNs' in expected:expected['mtime_ns']=expected.pop('mtimeNs')
        try:
            actual=item(name)
            for k,v in expected.items():
                if actual.get(k)!=v:failures.append({'path':name,'key':k,'before':v,'after':actual.get(k)})
        except Exception as e:failures.append({'path':name,'error':repr(e)})
    return failures

def digest(p):
    h=hashlib.sha256()
    with Path(p).open('rb') as f:
        for chunk in iter(lambda:f.read(1048576),b''): h.update(chunk)
    return h.hexdigest()
def load(p): return json.loads(Path(p).read_text())
def save(p,d):

def item(p):
    p=Path(p);s=p.lstat()
    if stat.S_ISLNK(s.st_mode):return {'kind':'link','target':os.readlink(p),'mtime_ns':s.st_mtime_ns}
    assert stat.S_ISREG(s.st_mode),str(p)
    return {'kind':'file','size':s.st_size,'mtime_ns':s.st_mtime_ns,'sha256':digest(p)}
def verify_old(records):
    failures=[]
    for name,old in records.items():
        try:
            actual=item(name)
            old=dict(old)
            if 'mtimeNs' in old:old['mtime_ns']=old.pop('mtimeNs')
            for k,v in old.items():
                if actual.get(k)!=v:failures.append({'path':name,'key':k,'expected':v,'actual':actual.get(k)})
        except Exception as exc: failures.append({'path':name,'error':repr(exc)})
    return failures

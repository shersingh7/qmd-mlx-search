def sha256(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b''):
            h.update(chunk)
    return h.hexdigest()


def dump(path, value):
    Path(path).write_text(json.dumps(value, indent=2, sort_keys=True))


def inventory(root):
    root = Path(root)
    entries = {}
    counts = {'file': 0, 'directory': 0, 'symlink': 0}
    regular_bytes = 0
    for directory, dirs, files in os.walk(root, followlinks=False):
        for name in sorted(dirs + files):
            p = Path(directory) / name
            s = p.lstat()
            entry = {'mode': stat.S_IMODE(s.st_mode)}
            if stat.S_ISLNK(s.st_mode):
                entry.update(kind='symlink', target=os.readlink(p))
            elif stat.S_ISDIR(s.st_mode):
                entry.update(kind='directory')
            elif stat.S_ISREG(s.st_mode):
                entry.update(kind='file', bytes=s.st_size, sha256=sha256(p))
                regular_bytes += s.st_size
            else:
                raise AssertionError(('Unexpected nonregular entry', str(p)))
            counts[entry['kind']] += 1
            entries[str(p.relative_to(root))] = entry
    return {'counts': counts, 'regular_file_bytes': regular_bytes, 'entries': entries}


def state_digest(inv):
    return hashlib.sha256(json.dumps(inv, sort_keys=True, separators=(',', ':')).encode()).hexdigest()


def split_header(data):
    matches = list(re.finditer(rb'(?m)^# /// script\r?\n(?:[^\n]*\n)*?^# ///\r?\n', data))
    assert len(matches) == 1
    m = matches[0]
    return data[:m.start()], data[m.start():m.end()], data[m.end():]


def originals():

    paths = set(intake['protected']) | {str(INTAKE)}
    paths.update(str(PREVIEW / 'repo' / rel) for rel in preview_inputs['inputs_sha256'])
    paths.update(str(REVIEW / 'mirror' / rel) for rel in intake['union_paths'])
    paths.update(str(PREVIEW / name) for name in ('input-manifest.json', 'preview-summary.json', 'cold-project-install.json', 'cold-project-environment-inventory.json'))
    return {'files': {str(p): sha256(p) for p in sorted(paths)},
            'old_qualification_full_tree': inventory(OLD)}

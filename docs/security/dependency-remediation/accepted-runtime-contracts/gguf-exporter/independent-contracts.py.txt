def digest(path):
    h=hashlib.sha256()
    with Path(path).open('rb') as f:
        for block in iter(lambda:f.read(1024*1024),b''):h.update(block)
    return h.hexdigest()


@functools.lru_cache()
def writer_bytes(kind='f16',label='fresh',big=False,alignment=32):
    from gguf import GGMLQuantizationType as T, GGML_QUANT_SIZES, GGUFEndian, GGUFWriter, LlamaFileType as L
    path=OUT/'writer'/f'{kind}-{label}-{big}-{alignment}.gguf'
    path.parent.mkdir(exist_ok=True)
    w=GGUFWriter(path,'qwen3',endianess=GGUFEndian.BIG if big else GGUFEndian.LITTLE)
    typ,ft={'f16':(T.F16,L.MOSTLY_F16),'q4_k_m':(T.Q4_K,L.MOSTLY_Q4_K_M),
            'q6_k':(T.Q6_K,L.MOSTLY_Q6_K),'q8_0':(T.Q8_0,L.MOSTLY_Q8_0)}[kind]
    try:
        w.add_name('independent synthetic zero blocks '+label)
        w.add_custom_alignment(alignment);w.add_file_type(ft)
        # Two tensors, including an earlier small typed F32 value and a packed
        # quantized final tensor. Never construct a custom purported GGUF header.
        w.add_tensor('independent.scalar',np.zeros((8,),dtype=np.float32))
        if kind=='f16':w.add_tensor('independent.matrix',np.zeros((32,256),dtype=np.float16))
        else:
            _,sz=GGML_QUANT_SIZES[typ]
            # The last tensor is a multiple of all tested alignments, so its
            # final byte is actual tensor data, not optional tail padding.
            w.add_tensor('independent.matrix',np.zeros((128,sz),dtype=np.uint8),raw_dtype=typ)
        w.write_header_to_file();w.write_kv_data_to_file();w.write_tensors_to_file()
    finally:w.close()
    return path.read_bytes()


def read_scalars(path):
    from gguf import GGUFReader
    r=GGUFReader.__new__(GGUFReader)
    try:
        r.__init__(path,mode='r')
        return {'tensor_count':len(r.tensors),'tensor_types':[t.tensor_type.name for t in r.tensors],
                'data_offset':int(r.data_offset),'end':max((int(t.data_offset)+int(t.n_bytes) for t in r.tensors),default=0),
                'nbytes':int(r.data.nbytes),'file_type':int(r.get_field('general.file_type').parts[-1][0]),
                'architecture':r.get_field('general.architecture').contents(),
                'vocab':len(r.get_field('tokenizer.ggml.tokens').data) if r.get_field('tokenizer.ggml.tokens') else None}
    finally:
        a=getattr(r,'data',None)
        if a is not None:a._mmap.close()

@pytest.mark.parametrize('cut',[5,7,8,15,16,23,'metadata','tensor-info','last-tensor'])
def test_constructor_and_tensor_truncations_do_not_leak_or_retain_parser_frames(tmp_path,monkeypatch,cut):
    from gguf import GGUFReader
    content=writer_bytes('q8_0');p=tmp_path/'resource.gguf';p.write_bytes(content)
    r=GGUFReader(p,mode='r')
    try:
        n={'metadata':int(r.get_field('general.name').offset)+10,
           'tensor-info':int(r.tensors[1].field.offset)+10,'last-tensor':len(content)-1}.get(cut,cut)
    finally:r.data._mmap.close()
    p.write_bytes(content[:n])
    reader_module=importlib.import_module('gguf.gguf_reader');np_real=reader_module.np;seen=[]
    class Watch:
        def __getattr__(self,n):return getattr(np_real,n)
        def memmap(self,*a,**k):
            array=np_real.memmap(*a,**k);seen.append(array._mmap);return array
    monkeypatch.setattr(reader_module,'np',Watch())
    retained=[];before=len(os.listdir('/dev/fd'));was=gc.isenabled();gc.disable()
    try:
        for i in range(17):
            try:tools.validate_gguf(p)
            except RuntimeError as e:
                retained.append(e)
                assert e.__context__ is None and e.__cause__ is None
                for frame,_line in traceback.walk_tb(e.__traceback__):
                    if Path(frame.f_code.co_filename).name=='gguf_toolchain.py':
                        assert not any(isinstance(v,(np.ndarray,GGUFReader)) for v in frame.f_locals.values())
                # Rendering complete outward diagnostics with locals must not
                # dereference any abandoned, closed NumPy parse views.
                text=''.join(traceback.TracebackException.from_exception(e,capture_locals=False).format())
                helper_frames=[(f,l) for f,l in traceback.walk_tb(e.__traceback__)
                               if Path(f.f_code.co_filename).name=='gguf_toolchain.py']
                rendered_locals=traceback.StackSummary.extract(helper_frames,capture_locals=True).format()
                assert rendered_locals
                assert 'invalid GGUF artifact' in text
            else:pytest.fail('Genuine truncated writer output accepted')
            assert len(seen)==i+1 and all(m.closed for m in seen)
            assert len(os.listdir('/dev/fd'))==before
        assert len(retained)==17
    finally:
        if was:gc.enable()
    record('resources',{'cut':cut,'repeats':17,'mapping_count':len(seen),'all_closed':all(m.closed for m in seen),
                        'fd_before':before,'fd_after':len(os.listdir('/dev/fd')),'retained':len(retained),'gc_disabled':True})

def fake_boundary(monkeypatch,tmp_path,fail_stage=None,fail_mode=None):
    # Independent transport, intentionally not the product's regression fakes.
    events=[]
    def preflight(**kwargs):
        events.append(('preflight',kwargs))
        return {n:'/INDEPENDENT/'+n for n in ('git','cmake','cc','c++','make')}
    monkeypatch.setattr(tools,'check_prerequisites',preflight)
    def run(argv,**kwargs):
        a=list(map(str,argv));p=None;kind=None;stdout=''
        if a[0]=='/INDEPENDENT/git':
            stage=next(x for x in ('init','fetch','checkout','rev-parse') if x in a)
            assert kwargs['env']['GIT_TERMINAL_PROMPT']=='0' and kwargs['env']['GIT_CONFIG_NOSYSTEM']=='1'
            if stage=='fetch':assert a[-2:]==[tools.SOURCE_URL,tools.SOURCE_REVISION]
            if stage=='rev-parse':stdout=tools.SOURCE_REVISION+'\n'
        elif a[0]=='/INDEPENDENT/cmake':
            if '--build' in a:
                stage='build';binary=Path(a[2])/'bin/llama-quantize';binary.parent.mkdir(parents=True);binary.write_text('NEVER EXECUTED')
                assert a[-2:]==['-j','2']
            else:
                stage='configure';assert '-DGGML_CUDA=OFF' in a and '-DGGML_METAL=OFF' in a and '-DGGML_OPENMP=OFF' in a
        elif a[0].endswith('llama-quantize'):
            stage=a[-1];kind=stage.lower();p=Path(a[2])
        elif any(x.endswith('convert_hf_to_gguf.py') for x in a):
            assert a[:2]==[sys.executable,'-s']
            stage='probe' if '--help' in a else 'converter'
            assert not set(kwargs['env'])&{'PYTHONPATH','PYTHONHOME','NO_LOCAL_GGUF'}
            if stage=='converter':kind='f16';p=Path(a[a.index('--outfile')+1])
        elif len(a)>1 and a[1].endswith('eval.py'):
            events.append(('eval',));return NS(returncode=0)
        else:raise AssertionError(a)
        events.append(('process',stage))
        if p is not None:
            b=writer_bytes(kind,label='fresh')
            p.write_bytes(b[:-1] if stage==fail_stage and fail_mode=='truncated' else b)
        return NS(returncode=17 if stage==fail_stage and fail_mode=='nonzero-valid' else 0,stdout=stdout,stderr='independent status diagnostic')
    monkeypatch.setattr(subprocess,'run',run)
    def save(path,**kwargs):
        Path(path).mkdir(parents=True,exist_ok=True);events.append(('save',str(path)))
    model=NS(save_pretrained=save)
    model.merge_and_unload=lambda:(events.append(('merge',)) or model)
    model.push_to_hub=lambda repo:events.append(('adapter-upload',repo))
    tok=NS(pad_token=None,eos_token='END',save_pretrained=save,push_to_hub=lambda repo:events.append(('tokenizer-upload',repo)))
    import datasets, transformers, trl, huggingface_hub
    monkeypatch.setattr(transformers.AutoModelForCausalLM,'from_pretrained',lambda *a,**k:(events.append(('model-load',)) or model))
    monkeypatch.setattr(transformers.AutoTokenizer,'from_pretrained',lambda *a,**k:(events.append(('tokenizer-load',)) or tok))
    import peft
    monkeypatch.setattr(peft.PeftModel,'from_pretrained',lambda *a,**k:model)
    class Data:
        def shuffle(self,**k):return self
        def train_test_split(self,**k):return {'train':self,'test':self}
    class Trainer:
        def __init__(self,**k):self.model=model
        def train(self):events.append(('training-fake',))
        def save_model(self,path):save(path)
    monkeypatch.setattr(datasets,'load_dataset',lambda *a,**k:Data())
    monkeypatch.setattr(trl,'SFTTrainer',Trainer)
    class Api:
        def create_repo(self,**k):events.append(('create-repo',))
        def upload_file(self,**k):events.append(('gguf-upload' if k['path_in_repo'].endswith('.gguf') else 'readme-upload',k['path_in_repo']))
    monkeypatch.setattr(huggingface_hub,'HfApi',Api)
    import convert_gguf as cli
    monkeypatch.setattr(cli,'HfApi',Api)
    monkeypatch.setattr(cli,'login',lambda **k:pytest.fail('No auth permitted'))
    monkeypatch.setattr(cli.tempfile,'gettempdir',lambda:str(tmp_path))
    return events,model,tok


def invoke(caller,monkeypatch,tmp_path,model,tok,extra=()):
    import train, train_unsloth, convert_gguf
    if caller=='shared':return train.export_gguf(model,tok,str(tmp_path/'adapter'),'adapter')
    if caller=='native-main':
        monkeypatch.setattr(sys,'argv',['train_unsloth.py','--model','0.8B','--device','cpu','--output',str(tmp_path/'adapter'),
                                       '--data',str(tmp_path/'never-read.jsonl'),'--push-hub','fixture/no-write',*extra])
        return train_unsloth.main()
    monkeypatch.setattr(sys,'argv',['convert_gguf.py','--size','1.7B','--output','fixture/adapter-gguf',*extra])
    return convert_gguf.main()


@pytest.mark.parametrize('caller',['shared','native-main','standalone'])
@pytest.mark.parametrize('stage',['converter','Q4_K_M','Q6_K','Q8_0'])
@pytest.mark.parametrize('failure',['truncated','nonzero-valid'])
def test_independent_public_entrypoints_refuse_without_prior_output_loss(tmp_path,monkeypatch,capsys,caller,stage,failure):
    events,model,tok=fake_boundary(monkeypatch,tmp_path,stage,failure)
    dest=tmp_path/('qmd-gguf-output-previous' if caller=='standalone' else 'adapter/gguf');dest.mkdir(parents=True)
    previous={q:writer_bytes(q,label='old') for q in ('q4_k_m','q6_k','q8_0')}
    for q,b in previous.items():(dest/f'adapter-{q}.gguf').write_bytes(b)
    with pytest.raises(RuntimeError,match='GGUF export failed'):
        invoke(caller,monkeypatch,tmp_path,model,tok)
    assert all((dest/f'adapter-{q}.gguf').read_bytes()==b for q,b in previous.items())
    assert not any(e[0] in ('gguf-upload','eval','create-repo') for e in events)
    assert [e for e in events if e[0]=='process'][-1]==('process',stage)
    assert not list(tmp_path.glob('qmd-gguf-convert-*')) and not list((tmp_path/'adapter').glob('.gguf-*'))
    if caller=='standalone':assert list(tmp_path.glob('qmd-gguf-output-*'))==[dest]
    output=capsys.readouterr().out
    assert 'Done!' not in output and 'GGUF files saved to:' not in output
    if caller=='native-main':assert ('adapter-upload','fixture/no-write') in events and ('tokenizer-upload','fixture/no-write') in events
    record('entrypoints',{'caller':caller,'stage':stage,'failure':failure,'previous_set_preserved':True,'events':events})


@pytest.mark.parametrize('caller',['shared','native-main','standalone'])
def test_independent_public_entrypoint_positive_complete_set(tmp_path,monkeypatch,capsys,caller):
    events,model,tok=fake_boundary(monkeypatch,tmp_path)
    invoke(caller,monkeypatch,tmp_path,model,tok)
    if caller=='standalone':
        destinations=list(tmp_path.glob('qmd-gguf-output-*'));assert len(destinations)==1;dest=destinations[0]
        expected=('f16','q4_k_m','q6_k','q8_0')
    else:dest=tmp_path/'adapter/gguf';expected=('q4_k_m','q6_k','q8_0')

    assert all((dest/f'adapter-{q}.gguf').read_bytes()==writer_bytes(q,label='fresh') for q in expected)
    assert 'GGUF files saved to:' in capsys.readouterr().out
    processes=[e[1] for e in events if e[0]=='process'];assert processes==['init','fetch','checkout','rev-parse','probe','configure','build','converter','Q4_K_M','Q6_K','Q8_0']
    record('entrypoints',{'caller':caller,'positive':True,'expected_files':list(expected),'events':events})
